//handler:

type commonHandler struct {
	writer io.Writer
	opts   *slog.HandlerOptions
	groups map[string]struct{}
	attrs  []slog.Attr
}

func (ch *commonHandler) clone() *commonHandler {
	// We can't use assignment because we can't copy the mutex.
	newGroups := make(map[string]struct{})
	for g := range ch.groups {
		newGroups[g] = struct{}{}
	}

	return &commonHandler{
		writer: ch.writer,
		opts:   ch.opts,
		attrs:  slices.Clip(ch.attrs),
		groups: newGroups,
	}
}

func (ch *commonHandler) withGroup(name string) *commonHandler {
	ch2 := ch.clone()

	ch2.groups[name] = struct{}{}

	return ch2
}

func (ch *commonHandler) writeTag(ctx context.Context, builder *bufio.Writer) {
	// Открываем теги
	_, _ = builder.WriteString("\"t\":[")

	var index int
	var group string
	for group = range ch.groups {
		if index == 0 {
			index++
		} else {
			_ = builder.WriteByte(44)
		}

		_ = builder.WriteByte(34)
		_, _ = builder.WriteString(group)
		_ = builder.WriteByte(34)
	}

	for group = range GetGroupsFromContext(ctx) {
		_, found := ch.groups[group] //slices.BinarySearch(ch.groups, group)
		if !found {
		if index == 0 {
			index++
		} else {
			_ = builder.WriteByte(44)
		}

			_ = builder.WriteByte(34)
			_, _ = builder.WriteString(group)
			_ = builder.WriteByte(34)
		}
	}

	// закрываем теги
	_, _ = builder.WriteString("],")
}

func GetGroupsFromContext(ctx context.Context) map[string]struct{} {
	contextGroups, _ := ctx.Value(KeyContextGroups).(map[string]struct{})

	return contextGroups
}

//context

func ContextWithGroups(ctx context.Context, groups ...string) context.Context {
	contextGroups := GetGroupsFromContext(ctx)

	if contextGroups == nil {
		contextGroups = make(map[string]struct{})
	}

	// здесь удаляются повторы
	for _, grp := range groups {
		contextGroups[grp] = struct{}{}
	}

	return context.WithValue(ctx, KeyContextGroups, contextGroups)
}
